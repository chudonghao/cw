if(NOT CLANG_EXECUTABLE)
  message(STATUS "Clang IR reference targets disabled: the project LLVM's Clang is required")
  return()
endif()

add_custom_target(codegen_case_clang_ir_all)

function(add_codegen_case_clang_ir name)
  set(target_name "codegen_case_clang_ir_${name}")
  add_custom_target(${target_name}
    COMMAND
      "${CMAKE_COMMAND}"
      "-DCLANG_EXECUTABLE=${CLANG_EXECUTABLE}"
      "-DTARGET_TRIPLE=${LLVM_HOST_TRIPLE}"
      "-DCASE_DIRECTORY=${CMAKE_CURRENT_SOURCE_DIR}"
      "-DCASE_NAME=${name}"
      "-DCASE_OPTIONS=${ARGN}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/generate_clang_ir.cmake"
    COMMENT "Generate Clang IR for ${name}"
    VERBATIM
  )
  add_dependencies(codegen_case_clang_ir_all ${target_name})
endfunction()
